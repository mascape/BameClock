<script>
/* =========================================
   STORAGE
========================================= */

const STORAGE_KEY = "basketball_game_clock_v1";

/* =========================================
   STATE
========================================= */

const state = {

  mode:"READY",

  gameMinutes:6,

  breakMinutes:2,

  gameCount:1,

  remaining:360,

  running:false,

  timerId:null,

  lastTick:0,

  warning60Played:false,

  warning15Played:false,

  wakeLock:null
};

/* =========================================
   ELEMENTS
========================================= */

const timerDisplay =
  document.getElementById("timerDisplay");

const modeLabel =
  document.getElementById("modeLabel");

const gameCountLabel =
  document.getElementById("gameCountLabel");

const pauseBtn =
  document.getElementById("pauseBtn");

const resetBtn =
  document.getElementById("resetBtn");

const setupPanel =
  document.getElementById("setupPanel");

const rotateMessage =
  document.getElementById("rotateMessage");

const mainScreen =
  document.getElementById("mainScreen");

const warning1 =
  document.getElementById("warning1");

const warning2 =
  document.getElementById("warning2");

const finish =
  document.getElementById("finish");

/* =========================================
   SAVE / LOAD
========================================= */

function saveSettings(){

  localStorage.setItem(
    STORAGE_KEY,

    JSON.stringify({

      gameMinutes:state.gameMinutes,

      breakMinutes:state.breakMinutes,

      gameCount:state.gameCount
    })
  );
}

function loadSettings(){

  const data =
    localStorage.getItem(STORAGE_KEY);

  if(!data) return;

  const parsed = JSON.parse(data);

  state.gameMinutes =
    parsed.gameMinutes || 6;

  state.breakMinutes =
    parsed.breakMinutes || 2;

  state.gameCount =
    parsed.gameCount || 1;

  updateOptionUI();
}

/* =========================================
   OPTION UI
========================================= */

function updateOptionUI(){

  document
    .querySelectorAll("[data-game]")
    .forEach(el=>{

      el.classList.remove("active");

      if(
        Number(el.dataset.game)
        ===
        state.gameMinutes
      ){
        el.classList.add("active");
      }
    });

  document
    .querySelectorAll("[data-break]")
    .forEach(el=>{

      el.classList.remove("active");

      if(
        Number(el.dataset.break)
        ===
        state.breakMinutes
      ){
        el.classList.add("active");
      }
    });

  gameCountLabel.textContent =
    `GAME ${state.gameCount}`;

  if(state.mode === "READY"){

    timerDisplay.textContent =
      formatMainTime(
        state.gameMinutes * 60
      );
  }
}

/* =========================================
   FORMAT
========================================= */

function formatMainTime(sec){

  if(sec <= 60){

    return sec.toFixed(1);
  }

  const m =
    Math.floor(sec / 60);

  const s =
    Math.floor(sec % 60);

  return `${m}:${String(s).padStart(2,"0")}`;
}

/* =========================================
   DISPLAY
========================================= */

function updateDisplay(){

  timerDisplay.className = "timer";

  if(state.remaining <= 60){

    timerDisplay.classList.add("large");
  }

  if(state.mode === "BREAK"){

    document.body.classList.add("break-mode");

  }else{

    document.body.classList.remove("break-mode");
  }

  if(state.mode === "GAME"){

    if(state.remaining <= 60){

      timerDisplay.classList.add("warning");
    }

    if(state.remaining <= 15){

      timerDisplay.classList.remove("warning");

      timerDisplay.classList.add("danger");
    }
  }

  timerDisplay.textContent =
    formatMainTime(state.remaining);
}

/* =========================================
   START GAME
========================================= */

function startGame(){

  state.mode = "GAME";

  state.remaining =
    state.gameMinutes * 60;

  state.warning60Played = false;

  state.warning15Played = false;
  
  modeLabel.textContent = "GAME";

  modeLabel.style.display = "none";

  setupPanel.classList.add("hidden");

  pauseBtn.classList.remove("hidden");

  resetBtn.classList.remove("hidden");

  pauseBtn.textContent = "PAUSE";

  runTimer();
}

/* =========================================
   START BREAK
========================================= */

function startBreak(){

  state.mode = "BREAK";

  state.remaining =
    state.breakMinutes * 60;

  modeLabel.textContent = "BREAK";

  modeLabel.style.display = "none";

  runTimer();
}

/* =========================================
   TIMER
========================================= */

function runTimer(){

  clearInterval(state.timerId);

  state.running = true;

  state.lastTick = performance.now();

  state.timerId = setInterval(()=>{

    const now = performance.now();

    const delta =
      (now - state.lastTick) / 1000;

    state.lastTick = now;

    state.remaining -= delta;

    /* 60 SEC */

    if(
      state.mode === "GAME"
      &&
      state.remaining <= 60
      &&
      !state.warning60Played
    ){

      warning1.currentTime = 0;

      warning1.play();

      state.warning60Played = true;
    }

    /* 15 SEC */

    if(
      state.mode === "GAME"
      &&
      state.remaining <= 15
      &&
      !state.warning15Played
    ){

      warning2.currentTime = 0;

      warning2.play();

      state.warning15Played = true;
    }

    /* FINISH */

    if(state.remaining <= 0){

      state.remaining = 0;

      updateDisplay();

      clearInterval(state.timerId);

      state.running = false;

      handleFinish();

      return;
    }

    updateDisplay();

  },100);
}

/* =========================================
   FINISH
========================================= */

function handleFinish(){

  finish.currentTime = 0;

  finish.play();

  /* GAME END */

  if(state.mode === "GAME"){

    state.gameCount++;

    saveSettings();

    modeLabel.textContent = "GAME SET";

    document.body.classList.add("flash-red");

    timerDisplay.textContent = "0.0";

    setTimeout(()=>{

      document.body.classList.remove("flash-red");

      gameCountLabel.textContent =
        `GAME ${state.gameCount}`;

      startBreak();

    },2000);
  }

  /* BREAK END */

  else if(state.mode === "BREAK"){

    modeLabel.textContent = "READY?";

    document.body.classList.add("flash-blue");

    timerDisplay.textContent = "0.0";

    setTimeout(()=>{

      document.body.classList.remove("flash-blue");

      returnReady();

    },2000);
  }
}

/* =========================================
   READY
========================================= */

function returnReady(){

  state.mode = "READY";

  clearInterval(state.timerId);

  state.running = false;

  pauseBtn.classList.add("hidden");

  resetBtn.classList.add("hidden");

  setupPanel.classList.remove("hidden");

  modeLabel.style.display = "block";
  modeLabel.textContent = "READY";

  state.remaining =
    state.gameMinutes * 60;

  updateDisplay();

  saveSettings();
}

/* =========================================
   BUTTON EVENTS
========================================= */

/* PAUSE */

pauseBtn.addEventListener("click",()=>{

  if(!state.running){

    runTimer();

    pauseBtn.textContent = "PAUSE";

    return;
  }

  clearInterval(state.timerId);

  state.running = false;

  pauseBtn.textContent = "RESUME";
});

/* RESET */

resetBtn.addEventListener("click",()=>{

  const ok =
    confirm("READYへ戻しますか？");

  if(!ok) return;

  returnReady();

  pauseBtn.textContent = "PAUSE";
});

/* START */

document
  .getElementById("startBtn")
  .addEventListener("click", async ()=>{

    try{

      if("wakeLock" in navigator){

        state.wakeLock =
          await navigator.wakeLock
          .request("screen");
      }

    }catch(e){}

    startGame();
  });

/* COUNT RESET */

document
  .getElementById("countResetBtn")
  .addEventListener("click",()=>{

    const ok =
      confirm(
        "GAME COUNT をリセットしますか？"
      );

    if(!ok) return;

    state.gameCount = 1;

    saveSettings();

    updateOptionUI();
  });

/* GAME TIME */

document
  .querySelectorAll("[data-game]")
  .forEach(el=>{

    el.addEventListener("click",()=>{

      state.gameMinutes =
        Number(el.dataset.game);

      saveSettings();

      updateOptionUI();
    });
  });

/* BREAK TIME */

document
  .querySelectorAll("[data-break]")
  .forEach(el=>{

    el.addEventListener("click",()=>{

      state.breakMinutes =
        Number(el.dataset.break);

      saveSettings();

      updateOptionUI();
    });
  });

/* =========================================
   ORIENTATION
========================================= */

function checkOrientation(){

  const portrait =
    window.innerHeight
    >
    window.innerWidth;

  if(portrait){

    rotateMessage.style.display = "flex";

    mainScreen.style.display = "none";

  }else{

    rotateMessage.style.display = "none";

    mainScreen.style.display = "flex";
  }
}

window.addEventListener(
  "resize",
  checkOrientation
);

/* =========================================
   INIT
========================================= */

loadSettings();

state.remaining =
  state.gameMinutes * 60;

updateDisplay();

checkOrientation();
</script>
